                              # Gibt allerdings nur den DNS-SD-Dienstnamen aus; für die
                              # Daten, die man wirklich braucht, s. cups-ipptool.
cups-ipptool                  # ippfind + ipptool. DAS Diagnose-Werkzeug hier: ippfind

                              # Damit steht VOR dem lpadmin fest, ob es klappt — wichtig,

avahi-tools                   # avahi-browse/avahi-resolve-host-name. Eigenes Paket, kommt
                              # NICHT mit avahi mit — und avahi ist die Schicht, auf die
                              # man ausweicht, wenn schon ippfind nichts findet.
avahi                         # avahi-daemon. Lag bisher nur als Abhängigkeit herum, jetzt
                              # als gewollter Dienst gelistet: CUPS und sane-airscan finden

                              # `lpinfo -v` einfach leer — keine Fehlermeldung.
nss-mdns                      # .local-Auflösung für glibc. ACHTUNG: das Paket liefert NUR

                              # trotzdem wirkungslos, bis bootstrap.sh es per
                              # `authselect enable-feature with-mdns4` verdrahtet.

                              # Das GUI-Frontend ist ein Flatpak (s. flatpaks.txt) und

#   hplip / gutenprint / foomatic-db — Treiberpakete. Ein IPP-Everywhere-Gerät meldet
